1"""Background task that prefetches today's exchange rate at 08:00 Belgrade time 

2(when kurs.resenje.org publishes NBS rates); retries every 30 minutes on failure. 

3See ``specs/reference/architecture.md`` for why this write is load-bearing beyond 

4the rate itself.""" 

5 

6import asyncio 

7import logging 

8from datetime import date, datetime, timedelta 

9from decimal import Decimal 

10from zoneinfo import ZoneInfo 

11 

12from dinary.adapters.rates.helpers import get_db_rate 

13from dinary.adapters.rates.service import get_rate 

14from dinary.config import settings 

15from dinary.db import storage 

16 

17logger = logging.getLogger(__name__) 

18 

19_BELGRADE = ZoneInfo("Europe/Belgrade") 

20_PREFETCH_HOUR = 8 

21_RETRY_INTERVAL_SEC = 1800 # 30 minutes 

22 

23 

24def _seconds_until_prefetch_hour() -> float: 

25 """Seconds until the next _PREFETCH_HOUR Belgrade time; DST-correct via ZoneInfo.""" 

26 now = datetime.now(tz=_BELGRADE) 

27 target = now.replace(hour=_PREFETCH_HOUR, minute=0, second=0, microsecond=0) 

28 if target <= now: 

29 target += timedelta(days=1) 

30 return max((target - now).total_seconds(), _RETRY_INTERVAL_SEC) 

31 

32 

33def _get_rate_blocking(rate_date: date, source: str, target: str) -> Decimal: 

34 """Runs in a worker thread; opens its own connection since sharing a SQLite 

35 handle across threads causes CPython 3.14 access violations.""" 

36 with storage.connection() as con: 

37 return get_rate(con, rate_date, source, target) 

38 

39 

40async def rate_prefetch_task() -> None: 

41 source = settings.app_currency 

42 target = settings.accounting_currency 

43 

44 while True: 

45 try: 

46 now_belgrade = datetime.now(tz=_BELGRADE) 

47 today = now_belgrade.date() 

48 

49 if now_belgrade.hour < _PREFETCH_HOUR: 

50 await asyncio.sleep(_seconds_until_prefetch_hour()) 

51 continue 

52 

53 with storage.connection() as con: 

54 already_stored = get_db_rate(con, today, source, target) is not None 

55 

56 if already_stored: 

57 logger.debug("rate for %s already stored", today) 

58 await asyncio.sleep(_seconds_until_prefetch_hour()) 

59 continue 

60 

61 rate = await asyncio.to_thread(_get_rate_blocking, today, source, target) 

62 

63 with storage.connection() as con: 

64 stored_now = get_db_rate(con, today, source, target) is not None 

65 

66 if not stored_now: 

67 # get_rate returned a stale fallback without writing today's rate. 

68 # Retry later so the daily write still happens once upstream recovers. 

69 logger.warning( 

70 "rate for %s/%s on %s not written to DB" 

71 " (stale fallback %s), retrying in %d min", 

72 source, 

73 target, 

74 today, 

75 rate, 

76 _RETRY_INTERVAL_SEC // 60, 

77 ) 

78 else: 

79 logger.info( 

80 "prefetched %s/%s rate for %s: %s", 

81 source, 

82 target, 

83 today, 

84 rate, 

85 ) 

86 await asyncio.sleep(_seconds_until_prefetch_hour()) 

87 continue 

88 except asyncio.CancelledError: 

89 raise 

90 except Exception: 

91 logger.exception("rate prefetch failed") 

92 

93 await asyncio.sleep(_RETRY_INTERVAL_SEC)
