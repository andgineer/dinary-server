15async def warm_sheet_mapping() -> None: 

16 """Moves the ~1s Drive+Sheets round-trip off the first-expense hot path. Bounded 

17 by ``warm_sheet_mapping_timeout_sec`` so a slow Google backend can't delay 

18 ``/api/health`` and starve the platform's health probe; on timeout/failure it 

19 logs and continues, since the drain loop retries on its own schedule. 

20 ``asyncio.wait_for`` cancels the awaitable but not the underlying ``to_thread`` 

21 worker, which keeps running until its socket timeout fires.""" 

22 if settings.sheet_logging_drain_interval_sec <= 0: 

23 return 

24 if not settings.sheet_logging_enabled: 

25 return 

26 timeout = settings.warm_sheet_mapping_timeout_sec 

27 if timeout <= 0: 

28 return 

29 try: 

30 summary = await asyncio.wait_for( 

31 asyncio.to_thread(sheet_mapping.reload_now), 

32 timeout=timeout, 

33 ) 

34 logger.info("sheet_mapping preloaded at startup: %s", summary) 

35 except TimeoutError: 

36 logger.warning( 

37 "sheet_mapping preload timed out after %.1fs; " 

38 "drain loop will retry on its own schedule", 

39 timeout, 

40 ) 

41 except Exception: 

42 logger.exception("sheet_mapping preload failed; drain loop will retry")

45async def sheet_logging_task() -> None: 

46 interval = settings.sheet_logging_drain_interval_sec 

47 enabled = settings.sheet_logging_enabled 

48 if not enabled or interval <= 0: 

49 logger.info( 

50 "sheet-logging task disabled (interval<=0 or DINARY_SHEET_LOGGING_SPREADSHEET unset)", 

51 ) 

52 return 

53 # Lets producers (e.g. POST /api/expenses) kick a sweep immediately; the 

54 # timer remains the fallback for crash-recovery sweeps. 

55 wake = asyncio.Event() 

56 loop = asyncio.get_running_loop() 

57 sheet_logging.register_wake_channel(wake, loop) 

58 logger.info("sheet-logging task started: interval=%gs", interval) 

59 try: 

60 first = True 

61 while True: 

62 if not first: 

63 with contextlib.suppress(TimeoutError): 

64 await asyncio.wait_for(wake.wait(), timeout=interval) 

65 wake.clear() 

66 first = False 

67 try: 

68 summary = await asyncio.to_thread(sheet_logging.drain_pending) 

69 income_summary = await asyncio.to_thread(drain_income_pending) 

70 attempted = summary.get("attempted", 0) 

71 cap_reached = summary.get("cap_reached", False) 

72 poisoned = summary.get("poisoned", 0) 

73 failed = summary.get("failed", 0) 

74 if attempted > 0 or cap_reached or poisoned > 0 or failed > 0: 

75 logger.info("drain sweep: %s", summary) 

76 else: 

77 logger.debug("drain sweep: %s", summary) 

78 if income_summary.get("attempted", 0) > 0 or income_summary.get("poisoned", 0) > 0: 

79 logger.info("income drain sweep: %s", income_summary) 

80 else: 

81 logger.debug("income drain sweep: %s", income_summary) 

82 except asyncio.CancelledError: 

83 raise 

84 except Exception: 

85 logger.exception("drain sweep failed") 

86 finally: 

87 sheet_logging.clear_wake_channel()
